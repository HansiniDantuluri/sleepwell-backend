# 1. App.tsx  (or main.tsx) — Wrap entire app with AuthProvider
# ─────────────────────────────────────────────────────────────────────────────

# BEFORE (your current App.tsx):
  return <RouterProvider router={router} />;

# AFTER:
  import { AuthProvider } from "@/hooks/useAuth";

  return (
    <AuthProvider>
      <RouterProvider router={router} />
    </AuthProvider>
  );

  import { useAuth } from "@/hooks/useAuth";
  import { useNavigate } from "react-router";

  const { user, loading } = useAuth();
  const navigate = useNavigate();

  useEffect(() => {
    if (!loading) {
      // After the splash animation finishes:
      if (user) {
        navigate("/dashboard");          // already logged in
      } else {
        navigate("/onboarding");         // first time or logged out
      }
    }
  }, [user, loading]);

# 4. Dashboard.tsx — Load today's schedule + show current task
# ─────────────────────────────────────────────────────────────────────────────

  import { useSchedule } from "@/hooks/useSchedule";
  import { useAuth } from "@/hooks/useAuth";
  import { api } from "@/lib/api";

  const { user } = useAuth();
  const { schedule, loadForDate } = useSchedule();
  const today = new Date().toISOString().split("T")[0]; // "YYYY-MM-DD"

  useEffect(() => {
    loadForDate(today);
  }, [today]);

  // Map schedule.blocks to your existing timeline UI:
  // block.start_time, block.end_time, block.title, block.color, block.is_locked
  // block.is_auto_adjusted → show "Auto-adjusted" badge
  // block.block_type === "sleep" → show 🔒 locked

  // Current task: find block where current time falls between start_time and end_time
  const now = new Date();
  const currentBlock = schedule?.blocks.find(b => {
    const [sh, sm] = b.start_time.split(":").map(Number);
    const [eh, em] = b.end_time.split(":").map(Number);
    const blockStart = sh * 60 + sm;
    const blockEnd   = eh * 60 + em;
    const nowMins    = now.getHours() * 60 + now.getMinutes();
    return nowMins >= blockStart && nowMins < blockEnd;
  });

  // Sleep protection: check if bedtime block is intact
  const sleepBlock = schedule?.blocks.find(b => b.block_type === "sleep");
  const sleepSecured = !!sleepBlock?.is_locked;
